#include <stdio.h>
#include <stdlib.h>

int main()
{
int n ,i,highest_marks,lowest_marks,count,sum=0;
float pass_percentage;
//Input the No Of Students
    printf("Enter the Number Of Students: ");
    scanf("%d",&n);
int marks[n];
//Input the Marks
    for(i=0;i<n;i++){
        printf("Enter the Marks Of Student %d: ",i+1);
        scanf("%d",&marks[i]);
        if(marks[i] < 0 || marks[i] > 100){
        	printf("Invalid Marks!\n");
        	i--;
        	continue;
		}
		else{
			continue;
		}
   }
//Calculate the vlaues of Highest Lowest and Average
     for(i=0;i<n;i++){
            sum = sum + marks[i];
        if(i==0){
            highest_marks = marks[0];
            lowest_marks = marks[0];
            count = 0;
        }
        if(highest_marks < marks[i]){
            highest_marks = marks[i];
        }
        if(lowest_marks > marks[i]){
            lowest_marks = marks[i];
        }
        if(marks[i]>=50){
            count++;
        }
     }
     //print

     printf("Highest Marks: %d\nLowest Marks: %d\nAverage Marks: %d\nPassed Students: %d",highest_marks,lowest_marks,sum/n,count);
     pass_percentage = (count*100)/n;
     printf("\nPass Percentage: %.2f",pass_percentage);


 




    return 0;
}
