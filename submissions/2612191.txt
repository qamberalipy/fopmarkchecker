// switch case
switch (package_type){
	case 1: 
	price = data_amount*30;
	if(price >800){
		discount = 100;
		discounted_price = price-discount;
		printf("the discount is  rs %d", discount);
		printf("the final amount is rs %d", discounted_price);
	}
	break;
	
	case 2: 
	price = data_amount*25;
	if(price >800){
		discount = 100;
		discounted_price = price-discount;
		printf("the discount is  rs %d", discount);
		printf("the final amount is rs %d", discounted_price);}
	
	
	break;
	
	case 3: 
	price = data_amount*20;
	if(price >800){
		discount = 100;
		discounted_price = price-discount;
		printf("the discount is  rs %d", &discount);
		printf("the final amount is rs %d", discounted_price);}
	
	break;
	
	case 4: 
	charge = 50;
	price = data_amount*15+charge;
	printf("the final amount is %d\n", price);
	printf("the service charge is %d", &charge);
	
	break;
}
	return 0;
}
