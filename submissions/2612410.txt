#include <stdio.h>
#include <stdlib.h>


int main()
{
	int choice,rate,sc=0,gb,price,discount,fprice;
	
printf("please select mobile data pack from these:\n 1.Daily pack RS 30 per GB \n 2. Monthly pack RS 20 per GB \n 3. Weekly pack RS 25 per GB \n 4. Business pack RS 15 per GB ");
scanf("%d",&choice);

switch (choice)
{
	case 1:
		rate=30;
		break;
	
	case 2:
		rate=20;
		break;
	
	case 3:
		rate=25;
		break;
	case 4:
		rate=15;
		sc=50;
		break;
}
printf("please enter the GB you want");
scanf("%d",&gb);

price=rate*gb;
if (price>800) {
	discount=100;
}
fprice=price+sc-discount;
printf("-----your details---- \n");
printf("\nyour package type is : %d",choice);
printf("\nyour data in GBs is : %d",gb);
printf("\nyour price is :%d",price);
printf("\nyour discount is : %d",discount);
printf("\nyour service charges is : %d",sc);
printf("\nyour final payable amount is : %d",fprice);
return 0;
}
