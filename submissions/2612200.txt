#include <stdio.h>
#include <stdlib.h>

/* run this program using the console pauser or add your own getch, system("pause") or input loop */


int main(){
	int n,i,marks,highest,lowest,count,above50,average,sum;
	float percentage;
	printf("Enter number of students");
	scanf("%d",&n);
	
	for(i=1;i<=n;i++){
		printf("Enter marks of students");
		scanf("%d",&marks);
		
		if(marks<0||marks>100){
			printf("Marks can be between 0 and 100 ");
			continue;
		}
		
		if(count==0){
			highest=marks;
			lowest=marks;
		}
		
		if(marks>highest){
			highest=marks;
		}
		if(marks<lowest){
			lowest=marks;
		}
		
		if(marks>50){
			above50++;
		}
		sum=sum+marks;
		if(count>0){
			average=sum/count;
		}
		count++;
		
		percentage=(sum/n)*100;
		
		
		
	}
	printf("Highest marks= %d\n Lowest marks= %d\n Average marks = %d\n Number of students who passed above 50= %d\n percentage= %.2f\n",highest,lowest,average,above50,percentage);
}

int main(int argc, char *argv[]) {
	int package_type,price,discount,service_charges,final_amount,total,gb;
	char package_name[20];
	printf("Following are the packages\n1.Daily pack:Rs 30 per GB\n2.Weekly pack:Rs 25 per GB\n3.Monthly pack:Rs 20 per GB\n4.Business pack:Rs 15 per GB\n");
	printf("Select package type ");
	scanf("%d",&package_type);
	printf("Enter the amaount of data in gb ");
	scanf("%d",&gb);
	
	switch(package_type){
		case 1:
			price=30;
			sprintf(package_name,"Daily pack");
			break;
		case 2:
			price=25;
			sprintf(package_name,"Weekly pack");
			break;
		case 3:
			price=20;
			sprintf(package_name,"Monthly pack");
			break;
		case 4:
			price=15;
			sprintf(package_name,"Business pack");
			break;
		default:
			printf("Invalid selection");
			return 0;
			
			
	}
	
	total=gb*price;
	if(total>800){
		discount=total-100;
	}
	else{
		discount=0;
	}
	final_amount=total-discount;
	if(package_type==4){
		service_charges=final_amount+50;
		final_amount=service_charges;
	}
	else{
		final_amount=total-discount;
	}
	
	printf("Package type= %s\nData Amount =%d\nPrice=%d\nAfter Discount=%d\nService Charges=%d\nFinal Amount =%d\n",package_name,gb,total,discount,service_charges,final_amount);
	
	return 0;
}
