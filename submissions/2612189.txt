#include<stdio.h>
int main()
{
	int N, c, marks, high, low=100,sum=0,count,pr;
	printf("Enter the Number of Students:");
	scanf("%d",&N);
	for(c=1;c<=N;c++)
	{
		printf("Enter Marks of N Students (Out of 100):");
		scanf("%d",&marks);
		if(marks<0 || marks>100)
		{
			printf("Invalid Marks Inputted by the User.");
		}
		else
		{
			if(marks>=50)
			{
				count++;
			}
		        sum+=marks;
		}
		
		if(marks>high)
		{
			high=marks;
		}
		if(marks<low)
		{
			low=marks;
		}
	}
	printf("\nHighest Marks= %d",high);
	printf("\nLowest Marks= %d",low);
	printf("\nAverage Marks= %d",sum/N);
	printf("\nNo.of Students who passed (marks of 50 or more)= %d",count);
	pr=sum/N;
	printf("\nPass Percentage= %d%%",pr);
}

     bill=bill + s_charge;
