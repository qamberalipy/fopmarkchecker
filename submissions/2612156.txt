Question 1) #include <stdio.h>

int main() {
    int n;
    int passcount = 0;
    float totalmarks = 0;
    float highest = -1;   // decarling lowest 
    float lowest = 101;   // decarling highest
    float currentmarks;

    printf("Enter the number of students (N): ");
    if (scanf("%d", &n) != 1 || n <= 0) {
        printf("Invalid number of students. Exiting program.\n");
    }

    for (int i = 1; i <= n; i++) {
        printf("Enter marks for student %d (0-100): ", i);
        scanf("%f", &currentmarks);
        if (currentmarks < 0 || currentmarks > 100) { // not counting marks
            printf("Error: Invalid marks entered. Please enter a value between 0 and 100.\n");
            i--; //decreament of input
            continue;}
        totalmarks =totalmarks+ currentmarks;

        if (currentmarks > highest) { //finding high and lowest
            highest = currentmarks;
        }
        if (currentmarks < lowest) {
            lowest = currentmarks;
        }
        if (currentmarks >= 50) { //finding how many pass
            passcount++;
        }
    }
    float average = totalmarks/n;
    float passpercentage = ((float)passcount/n) * 100;
    printf(" Class Performance Statistics \n");
    printf("Total Students Processed: %d\n", n);
    printf("Highest Marks: %.2f\n", highest);
    printf("Lowest Marks: %.2f\n", lowest);
    printf("Average Marks: %.2f\n", average);
    printf("Number of Students Passed (>= 50): %d\n", passcount);
    printf("Pass Percentage of Class: %.2f%%\n", passpercentage);


}
