	//Question 1
	int n;
	int marks;
	int sum=0;
	int count=0;
	int highmark;
	int lowestmark;
	float percentage;
	float avg;
	printf("Enter the number of student u want to enter:");
	scanf("%d",&n);
	count++;
	
	printf("Enter the marks out of 100:");
	for(int i=1;i<=n;i++){
	scanf("%d",&marks);

	if(marks<0 || marks>100){
		printf("error");
		return 0;
	}else{
		sum=sum+i;
	if(marks>=50){
		printf("pass");
		avg=marks/count;
		printf("avg is:%.2f",avg);
	}
	if(marks>highestmark){
		printf("highest marks are:%d",marks);
		return 0;
	}
	if(marks<lowestmark){
		printf("lowest marks are:%d",marks);
	}
	percentage=sum/count;
	printf("percentage is:%.2f",percentage);
}
